import sys
# Add the custom framework directory to the system path to load shared modules
sys.path.append("/lakehouse/default/Files/framework")

# Import custom frameworks for validation checks and generic utility functions
from utils import *
from schema_utils import *

# Import PySpark functions and core tracking types
from pyspark.sql.functions import *
from pyspark.sql.types import *

# Generate unique operational identifiers to track this migration run in the audit trail
migration_pipeline_run_id = generate_pipeline_run_id()
source_system_identifier = "lims7_migration"

# Load the master configuration registry tracking your active pipelines
dataset_df = spark.table("cfg_lims_dataset").filter("active_flag = 'Y'")

print(f"Starting historical data migration run: {migration_pipeline_run_id}")
print("------------------------------------------------------------------")

# Sequentially loop through each active dataset definition to migrate records
for ds in dataset_df.collect():
    source_table_name = ds["source_table"]
    target_bronze_table = ds["bronze_table"]
    
    # Define your naming convention for the old legacy unpartitioned tables
    # Modify this string template if your legacy tables follow a different pattern
    legacy_unpartitioned_table = f"old_legacy_{source_table_name.lower()}"
    
    # Define a clean batch execution ID for this specific table loop
    migration_batch_id = generate_batch_id(source_system_identifier, source_table_name, "historical_migration")
    migration_start_time = utc_now()
    
    print(f"Migrating: {legacy_unpartitioned_table} ---> {target_bronze_table}")
    
    # Safety Check 1: Ensure the legacy unpartitioned source table actually exists
    if not table_exists(spark, legacy_unpartitioned_table):
        print(f"   [SKIP] Legacy source table '{legacy_unpartitioned_table}' not found.")
        continue
        
    # Safety Check 2: Ensure your newly optimized target structure is deployed
    if not table_exists(spark, target_bronze_table):
        print(f"   [SKIP] Optimized target bronze table '{target_bronze_table}' not found. Deploy DDL first.")
        continue

    try:
        # Step 1: Read the historical data from the unpartitioned legacy table
        legacy_df = spark.table(legacy_unpartitioned_table)
        total_records_read = legacy_df.count()
        
        if total_records_read == 0:
            print(f"   [INFO] Source table {legacy_unpartitioned_table} is empty. Skipping data sync.")
            continue
            
        # Step 2: Clean and filter columns to match your new table structure
        # Isolate target table columns, excluding any existing operational metadata fields
        target_schema_fields = spark.table(target_bronze_table).schema.names
        pure_business_columns = [c for c in target_schema_fields if not c.startswith("META_")]
        
        # Resolve case differences between legacy names and your new configuration definitions
        matched_source_columns = []
        for target_col in pure_business_columns:
            resolved_name = resolve_column_case_insensitive(legacy_df, target_col)
            if resolved_name:
                # Select the column and cast/alias it explicitly to match the new schema definition
                matched_source_columns.append(col(resolved_name).alias(target_col))
            else:
                # Handle schema drift: If a new column is missing in legacy data, fill it with a null literal
                target_type = [f.dataType for f in spark.table(target_bronze_table).schema.fields if f.name == target_col][0]
                matched_source_columns.append(lit(None).cast(target_type).alias(target_col))
                print(f"   [WARN] Column '{target_col}' missing in legacy table. Injected NULL fallback.")

        # Reconstruct the dataframe using only the safely matched business columns
        processed_legacy_df = legacy_df.select(*matched_source_columns)

        # Step 3: Inject your framework audit trail lineage and change-tracking hashes
        migrated_bronze_df = add_bronze_metadata(
            df=processed_legacy_df,
            batch_id=migration_batch_id,
            pipeline_run_id=migration_pipeline_run_id,
            source_system=source_system_identifier,
            source_table=source_table_name,
            load_type="full_historical_migration",
            watermark_value="HISTORICAL_MIGRATION"
        )

        # Step 4: Write records into your optimized, partitioned target storage structure
        print(f"   [WRITE] Appending {total_records_read} records into {target_bronze_table}...")
        
        # Use append mode to load data; mergeSchema handles any minor type evolution boundaries safely
        migrated_bronze_df.write.mode("append") \
            .option("mergeSchema", "true") \
            .saveAsTable(target_bronze_table)
            
        # Step 5: Run optimization and Microsoft Fabric VORDER file alignment
        print(f"   [OPTIMIZE] Consolidating parquet files and applying VORDER to {target_bronze_table}...")
        spark.sql(f"OPTIMIZE {target_bronze_table}")
        
        # Step 6: Log successful migration metrics into your framework audit trails
        append_log(spark, "ops_pipeline_run_log", {
            "pipeline_run_id": migration_pipeline_run_id,
            "batch_id": migration_batch_id,
            "source_system": source_system_identifier,
            "table_name": source_table_name,
            "landing_table": legacy_unpartitioned_table,
            "bronze_table": target_bronze_table,
            "load_type": "full_historical_migration",
            "start_time": migration_start_time,
            "end_time": utc_now(),
            "status": "SUCCESS",
            "records_read": total_records_read,
            "records_written": total_records_read,
            "records_rejected": 0,
            "error_message": None
        })
        print(f"   [SUCCESS] Migration completed smoothly for {target_bronze_table}.\n")

    except Exception as migration_error:
        error_string = str(migration_error)
        print(f"   [ERROR] Migration failed for table {source_table_name}: {error_string}\n")
        
        # Log the failure metrics to preserve your execution governance pipeline record
        append_log(spark, "ops_pipeline_run_log", {
            "pipeline_run_id": migration_pipeline_run_id,
            "batch_id": migration_batch_id,
            "source_system": source_system_identifier,
            "table_name": source_table_name,
            "landing_table": legacy_unpartitioned_table,
            "bronze_table": target_bronze_table,
            "load_type": "full_historical_migration",
            "start_time": migration_start_time,
            "end_time": utc_now(),
            "status": "FAILED",
            "records_read": 0,
            "records_written": 0,
            "records_rejected": 0,
            "error_message": error_string
        })

print("All active historical data migrations completed.")
